import 'package:flutter/material.dart';
import 'package:flutter_rating_bar/flutter_rating_bar.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:mockito/annotations.dart';
import 'package:mockito/mockito.dart';
import 'package:passenger_app/api/ride_service.dart';
import 'package:passenger_app/screens/rating_screen.dart';

import 'rating_screen_test.mocks.dart';

@GenerateMocks([RideService])
void main() {
  late MockRideService mockRideService;

  setUp(() {
    mockRideService = MockRideService();
  });

  Widget createWidgetUnderTest() {
    return MaterialApp(
      home: RatingScreen(
        rideService: mockRideService,
        tripId: 'trip-123',
        counterpartName: 'John',
      ),
    );
  }

  testWidgets('RatingScreen displays counterpart name and allows rating', (widgetTester) async {
    // Arrange
    await widgetTester.pumpWidget(createWidgetUnderTest());

    // Assert
    expect(find.text('How was your trip with John?'), findsOneWidget);
    expect(find.byType(RatingBar), findsOneWidget);
    expect(find.byType(TextField), findsOneWidget);
  });

  testWidgets('submitting a rating calls the ride service and navigates away', (widgetTester) async {
    // Arrange
    when(mockRideService.submitRating(
      tripId: anyNamed('tripId'),
      rating: anyNamed('rating'),
      comment: anyNamed('comment'),
    )).thenAnswer((_) async => {});

    await widgetTester.pumpWidget(createWidgetUnderTest());

    // Act
    // Change rating from default 3.0 to 5.0
    await widgetTester.tap(find.byIcon(Icons.star).last);
    await widgetTester.enterText(find.byType(TextField), 'Great driver!');
    await widgetTester.tap(find.widgetWithText(ElevatedButton, 'Submit Rating'));
    await widgetTester.pump(); // Show loading indicator

    // Assert
    expect(find.byType(CircularProgressIndicator), findsOneWidget);

    // Act
    await widgetTester.pumpAndSettle(); // Wait for future to complete

    // Assert
    verify(mockRideService.submitRating(tripId: 'trip-123', rating: 5.0, comment: 'Great driver!')).called(1);
    // The screen should have been popped, so it should no longer be in the widget tree.
    expect(find.byType(RatingScreen), findsNothing);
  });
}
