import 'package:share_plus/share_plus.dart';

class ShareTripService {
  /// Shares trip details via the native platform share dialog.
  Future<void> shareTripDetails({
    required String driverName,
    required String vehiclePlate,
    required String currentLocUrl,
    required String passengerName,
    String? eta,
  }) async {
    final StringBuffer message = StringBuffer();
    message.writeln('🚗 CABI Trip Update');
    message.writeln('-------------------');
    message.writeln('Driver: $driverName ($vehiclePlate)');
    message.writeln('Passenger: $passengerName');
    if (eta != null) {
      message.writeln('ETA: $eta');
    }
    message.writeln('📍 Live Location: $currentLocUrl');
    message.writeln('\nTrack this ride for safety.');

    await Share.share(
      message.toString(),
      subject: 'CABI Trip Status: $driverName',
    );
  }
}
