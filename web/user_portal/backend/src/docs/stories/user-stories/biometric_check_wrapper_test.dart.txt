import 'package:driver_app/api/user_service.dart';
import 'package:driver_app/services/biometric_service.dart';
import 'package:driver_app/widgets/biometric_check_wrapper.dart';
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:mockito/annotations.dart';
import 'package:mockito/mockito.dart';

import 'biometric_check_wrapper_test.mocks.dart';

@GenerateMocks([UserService, BiometricService])
void main() {
  late MockUserService mockUserService;
  late MockBiometricService mockBiometricService;

  setUp(() {
    mockUserService = MockUserService();
    mockBiometricService = MockBiometricService();
  });

  Widget createWidgetUnderTest() {
    return MaterialApp(
      home: BiometricCheckWrapper(
        userService: mockUserService,
        biometricService: mockBiometricService,
        child: const Text('Child Widget'),
      ),
    );
  }

  testWidgets('does not show overlay if verification is not required',
      (widgetTester) async {
    when(mockUserService.getUserProfile())
        .thenAnswer((_) async => {'requiresBiometricCheck': false});

    await widgetTester.pumpWidget(createWidgetUnderTest());
    await widgetTester.pumpAndSettle();

    expect(find.text('Child Widget'), findsOneWidget);
    expect(find.byType(Card), findsNothing);
  });

  testWidgets('shows overlay and removes it on successful verification',
      (widgetTester) async {
    when(mockUserService.getUserProfile())
        .thenAnswer((_) async => {'requiresBiometricCheck': true});
    when(mockBiometricService.authenticate()).thenAnswer((_) async => true);
    when(mockUserService.postBiometricSuccess()).thenAnswer((_) async {});

    await widgetTester.pumpWidget(createWidgetUnderTest());
    await widgetTester.pump(); // Initial build

    // Overlay should be visible
    expect(find.text('Identity Verification Required'), findsOneWidget);
    expect(find.text('Child Widget'), findsOneWidget); // Child is still in the tree

    await widgetTester.pumpAndSettle(); // Let futures complete

    // Overlay should be gone
    expect(find.text('Identity Verification Required'), findsNothing);
    verify(mockUserService.postBiometricSuccess()).called(1);
  });

  testWidgets('re-prompts on first failure, then succeeds',
      (widgetTester) async {
    when(mockUserService.getUserProfile())
        .thenAnswer((_) async => {'requiresBiometricCheck': true});
    when(mockBiometricService.authenticate())
        .thenAnswer((_) async => false) // First call fails
        .thenAnswer((_) async => true); // Second call succeeds
    when(mockUserService.postBiometricSuccess()).thenAnswer((_) async {});

    await widgetTester.pumpWidget(createWidgetUnderTest());
    await widgetTester.pump();

    // First attempt
    expect(find.text('Identity Verification Required'), findsOneWidget);
    await widgetTester.pump(const Duration(seconds: 2)); // Wait for re-prompt delay

    // Second attempt
    await widgetTester.pumpAndSettle();

    expect(find.text('Identity Verification Required'), findsNothing);
    verify(mockBiometricService.authenticate()).called(2);
    verify(mockUserService.postBiometricSuccess()).called(1);
  });

  testWidgets('calls postBiometricFailure after 3 failed attempts',
      (widgetTester) async {
    when(mockUserService.getUserProfile())
        .thenAnswer((_) async => {'requiresBiometricCheck': true});
    when(mockBiometricService.authenticate()).thenAnswer((_) async => false);
    when(mockUserService.postBiometricFailure()).thenAnswer((_) async {});

    await widgetTester.pumpWidget(createWidgetUnderTest());
    await widgetTester.pump();

    // Attempt 1
    await widgetTester.pump(const Duration(seconds: 2));
    // Attempt 2
    await widgetTester.pump(const Duration(seconds: 2));
    // Attempt 3
    await widgetTester.pumpAndSettle();

    // Overlay is still visible, but failure was posted
    expect(find.text('Identity Verification Required'), findsOneWidget);
    verify(mockBiometricService.authenticate()).called(3);
    verify(mockUserService.postBiometricFailure()).called(1);
  });
}
