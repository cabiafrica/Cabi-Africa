import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:mockito/annotations.dart';
import 'package:mockito/mockito.dart';
import 'package:passenger_app/api/user_service.dart';
import 'package:passenger_app/screens/feedback_screen.dart';

import 'feedback_screen_test.mocks.dart';

@GenerateMocks([UserService])
void main() {
  late MockUserService mockUserService;

  setUp(() {
    mockUserService = MockUserService();
  });

  Widget createWidgetUnderTest() {
    return MaterialApp(
      home: FeedbackScreen(userService: mockUserService),
    );
  }

  final mockFeedbackData = [
    {'comment': 'Very friendly!', 'visibleDate': '2024-05-01T00:00:00Z'},
    {'comment': 'Clean car.', 'visibleDate': '2024-05-01T00:00:00Z'},
    {'comment': 'Great driver.', 'visibleDate': '2024-04-01T00:00:00Z'},
  ];

  testWidgets('shows loading indicator while fetching data', (widgetTester) async {
    // Arrange
    when(mockUserService.getFeedback()).thenAnswer((_) => Future.delayed(const Duration(seconds: 2)));

    // Act
    await widgetTester.pumpWidget(createWidgetUnderTest());
    await widgetTester.pump(); // Start the future

    // Assert
    expect(find.byType(CircularProgressIndicator), findsOneWidget);
  });

  testWidgets('shows error message if fetching data fails', (widgetTester) async {
    // Arrange
    when(mockUserService.getFeedback()).thenThrow(Exception('API Error'));

    // Act
    await widgetTester.pumpWidget(createWidgetUnderTest());
    await widgetTester.pumpAndSettle(); // Wait for future to complete

    // Assert
    expect(find.textContaining('Error:'), findsOneWidget);
  });

  testWidgets('displays feedback grouped by month on successful fetch', (widgetTester) async {
    // Arrange
    when(mockUserService.getFeedback()).thenAnswer((_) async => mockFeedbackData);

    // Act
    await widgetTester.pumpWidget(createWidgetUnderTest());
    await widgetTester.pumpAndSettle();

    // Assert
    // Check for month headers
    expect(find.text('Feedback for May 2024'), findsOneWidget);
    expect(find.text('Feedback for April 2024'), findsOneWidget);

    // Check for comments
    expect(find.textContaining('Very friendly!'), findsOneWidget);
    expect(find.textContaining('Clean car.'), findsOneWidget);
    expect(find.textContaining('Great driver.'), findsOneWidget);
  });
}
