import 'package:flutter/material.dart';
import 'package:intl/intl.dart';

class FeedbackItem {
  final String content;
  final DateTime date;

  FeedbackItem({required this.content, required this.date});
}

class MyFeedbackScreen extends StatefulWidget {
  const MyFeedbackScreen({super.key});

  @override
  State<MyFeedbackScreen> createState() => _MyFeedbackScreenState();
}

class _MyFeedbackScreenState extends State<MyFeedbackScreen> {
  late Future<List<FeedbackItem>> _feedbackFuture;

  @override
  void initState() {
    super.initState();
    _loadFeedback();
  }

  void _loadFeedback() {
    // Simulate API call
    _feedbackFuture = Future.delayed(const Duration(seconds: 1), () {
      return [
        FeedbackItem(
          content: 'Great driver, very polite!',
          date: DateTime(2024, 5, 15),
        ),
        FeedbackItem(content: 'Car was spotless.', date: DateTime(2024, 5, 20)),
        FeedbackItem(content: 'Arrived on time.', date: DateTime(2024, 4, 10)),
      ];
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('My Feedback')),
      body: FutureBuilder<List<FeedbackItem>>(
        future: _feedbackFuture,
        builder: (context, snapshot) {
          if (snapshot.connectionState == ConnectionState.waiting) {
            return const Center(child: CircularProgressIndicator());
          }
          if (snapshot.hasError) {
            return Center(child: Text('Error: ${snapshot.error}'));
          }
          if (!snapshot.hasData || snapshot.data!.isEmpty) {
            return const Center(child: Text('No feedback available yet.'));
          }

          final feedbackList = snapshot.data!;
          // Group by month
          final groupedFeedback = <String, List<FeedbackItem>>{};
          for (var item in feedbackList) {
            final monthKey = DateFormat('MMMM yyyy').format(item.date);
            if (!groupedFeedback.containsKey(monthKey)) {
              groupedFeedback[monthKey] = [];
            }
            groupedFeedback[monthKey]!.add(item);
          }

          return ListView.builder(
            padding: const EdgeInsets.all(16),
            itemCount: groupedFeedback.length,
            itemBuilder: (context, index) {
              final monthKey = groupedFeedback.keys.elementAt(index);
              final items = groupedFeedback[monthKey]!;

              return Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Padding(
                    padding: const EdgeInsets.symmetric(vertical: 8.0),
                    child: Text(
                      'Feedback for $monthKey',
                      style: Theme.of(context).textTheme.titleMedium?.copyWith(
                        fontWeight: FontWeight.bold,
                        color: Colors.grey[700],
                      ),
                    ),
                  ),
                  ...items.map(
                    (item) => Card(
                      margin: const EdgeInsets.only(bottom: 8),
                      child: Padding(
                        padding: const EdgeInsets.all(16.0),
                        child: Text(
                          '"${item.content}"',
                          style: const TextStyle(fontStyle: FontStyle.italic),
                        ),
                      ),
                    ),
                  ),
                  const SizedBox(height: 16),
                ],
              );
            },
          );
        },
      ),
    );
  }
}
