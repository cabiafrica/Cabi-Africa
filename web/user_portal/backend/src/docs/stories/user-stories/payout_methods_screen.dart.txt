import 'package:flutter/material.dart';

enum PayoutType { mpesa, bank }

class PayoutMethod {
  final String id;
  final PayoutType type;
  final Map<String, String> details;
  bool isDefault;

  PayoutMethod({
    required this.id,
    required this.type,
    required this.details,
    this.isDefault = false,
  });
}

class PayoutMethodsScreen extends StatefulWidget {
  const PayoutMethodsScreen({Key? key}) : super(key: key);

  @override
  _PayoutMethodsScreenState createState() => _PayoutMethodsScreenState();
}

class _PayoutMethodsScreenState extends State<PayoutMethodsScreen> {
  List<PayoutMethod> methods = [];
  bool isLoading = true;

  @override
  void initState() {
    super.initState();
    _fetchMethods();
  }

  Future<void> _fetchMethods() async {
    // Simulate API fetch
    await Future.delayed(const Duration(seconds: 1));
    if (mounted) {
      setState(() {
        methods = [
          PayoutMethod(
            id: '1',
            type: PayoutType.mpesa,
            details: {'phone': '+254712345678'},
            isDefault: true,
          ),
        ];
        isLoading = false;
      });
    }
  }

  Future<void> _addMethod(PayoutType type, Map<String, String> details) async {
    setState(() => isLoading = true);
    // Simulate API call
    await Future.delayed(const Duration(seconds: 1));
    final newMethod = PayoutMethod(
      id: DateTime.now().millisecondsSinceEpoch.toString(),
      type: type,
      details: details,
      isDefault: methods.isEmpty, // Make default if it's the first one
    );
    setState(() {
      methods.add(newMethod);
      isLoading = false;
    });
    Navigator.pop(context); // Close dialog
  }

  Future<void> _setDefault(String id) async {
    setState(() {
      for (var m in methods) {
        m.isDefault = (m.id == id);
      }
    });
    // Simulate API sync
  }

  Future<void> _deleteMethod(String id) async {
    setState(() {
      methods.removeWhere((m) => m.id == id);
    });
    // Simulate API sync
  }

  void _showAddMethodDialog() {
    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      shape: const RoundedRectangleBorder(
        borderRadius: BorderRadius.vertical(top: Radius.circular(20)),
      ),
      builder: (context) => AddPayoutMethodForm(onSubmit: _addMethod),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Payout Methods'),
        backgroundColor: Colors.white,
        foregroundColor: Colors.black,
        elevation: 0,
      ),
      backgroundColor: Colors.grey[100],
      body: isLoading
          ? const Center(child: CircularProgressIndicator())
          : methods.isEmpty
          ? Center(
              child: Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  Icon(
                    Icons.account_balance_wallet_outlined,
                    size: 64,
                    color: Colors.grey[400],
                  ),
                  const SizedBox(height: 16),
                  Text(
                    'No payout methods added',
                    style: TextStyle(color: Colors.grey[600], fontSize: 16),
                  ),
                  const SizedBox(height: 24),
                  ElevatedButton(
                    onPressed: _showAddMethodDialog,
                    child: const Text('Add Payout Method'),
                  ),
                ],
              ),
            )
          : ListView.builder(
              padding: const EdgeInsets.all(16),
              itemCount: methods.length,
              itemBuilder: (context, index) {
                final method = methods[index];
                return Card(
                  margin: const EdgeInsets.only(bottom: 12),
                  shape: RoundedRectangleBorder(
                    borderRadius: BorderRadius.circular(12),
                  ),
                  child: ListTile(
                    leading: CircleAvatar(
                      backgroundColor: method.type == PayoutType.mpesa
                          ? Colors.green[50]
                          : Colors.blue[50],
                      child: Icon(
                        method.type == PayoutType.mpesa
                            ? Icons.phone_android
                            : Icons.account_balance,
                        color: method.type == PayoutType.mpesa
                            ? Colors.green
                            : Colors.blue,
                      ),
                    ),
                    title: Text(
                      method.type == PayoutType.mpesa
                          ? 'M-Pesa'
                          : method.details['bankName'] ?? 'Bank Transfer',
                    ),
                    subtitle: Text(
                      method.type == PayoutType.mpesa
                          ? method.details['phone'] ?? ''
                          : '${method.details['accountNumber']}',
                    ),
                    trailing: method.isDefault
                        ? const Icon(Icons.check_circle, color: Colors.green)
                        : PopupMenuButton(
                            onSelected: (value) {
                              if (value == 'default') {
                                _setDefault(method.id);
                              } else if (value == 'delete') {
                                _deleteMethod(method.id);
                              }
                            },
                            itemBuilder: (context) => [
                              const PopupMenuItem(
                                value: 'default',
                                child: Text('Set as Default'),
                              ),
                              const PopupMenuItem(
                                value: 'delete',
                                child: Text(
                                  'Delete',
                                  style: TextStyle(color: Colors.red),
                                ),
                              ),
                            ],
                          ),
                  ),
                );
              },
            ),
      floatingActionButton: methods.isNotEmpty
          ? FloatingActionButton(
              onPressed: _showAddMethodDialog,
              child: const Icon(Icons.add),
            )
          : null,
    );
  }
}

class AddPayoutMethodForm extends StatefulWidget {
  final Function(PayoutType, Map<String, String>) onSubmit;

  const AddPayoutMethodForm({Key? key, required this.onSubmit})
    : super(key: key);

  @override
  _AddPayoutMethodFormState createState() => _AddPayoutMethodFormState();
}

class _AddPayoutMethodFormState extends State<AddPayoutMethodForm> {
  PayoutType _selectedType = PayoutType.mpesa;
  final _formKey = GlobalKey<FormState>();
  final Map<String, String> _formData = {};

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: EdgeInsets.only(
        bottom: MediaQuery.of(context).viewInsets.bottom,
        left: 16,
        right: 16,
        top: 24,
      ),
      child: Form(
        key: _formKey,
        child: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            const Text(
              'Add Payout Method',
              style: TextStyle(fontSize: 20, fontWeight: FontWeight.bold),
            ),
            const SizedBox(height: 16),
            Row(
              children: [
                Expanded(
                  child: RadioListTile<PayoutType>(
                    title: const Text('M-Pesa'),
                    value: PayoutType.mpesa,
                    groupValue: _selectedType,
                    onChanged: (val) => setState(() => _selectedType = val!),
                    contentPadding: EdgeInsets.zero,
                  ),
                ),
                Expanded(
                  child: RadioListTile<PayoutType>(
                    title: const Text('Bank'),
                    value: PayoutType.bank,
                    groupValue: _selectedType,
                    onChanged: (val) => setState(() => _selectedType = val!),
                    contentPadding: EdgeInsets.zero,
                  ),
                ),
              ],
            ),
            const SizedBox(height: 16),
            if (_selectedType == PayoutType.mpesa) ...[
              TextFormField(
                decoration: const InputDecoration(
                  labelText: 'Phone Number',
                  border: OutlineInputBorder(),
                  prefixText: '+254 ',
                ),
                keyboardType: TextInputType.phone,
                validator: (value) {
                  if (value == null || value.isEmpty) return 'Required';
                  if (value.length < 9) return 'Invalid phone number';
                  return null;
                },
                onSaved: (val) => _formData['phone'] = '+254$val',
              ),
            ] else ...[
              TextFormField(
                decoration: const InputDecoration(
                  labelText: 'Bank Name',
                  border: OutlineInputBorder(),
                ),
                validator: (value) =>
                    value == null || value.isEmpty ? 'Required' : null,
                onSaved: (val) => _formData['bankName'] = val!,
              ),
              const SizedBox(height: 12),
              TextFormField(
                decoration: const InputDecoration(
                  labelText: 'Account Number',
                  border: OutlineInputBorder(),
                ),
                keyboardType: TextInputType.number,
                validator: (value) =>
                    value == null || value.isEmpty ? 'Required' : null,
                onSaved: (val) => _formData['accountNumber'] = val!,
              ),
              const SizedBox(height: 12),
              TextFormField(
                decoration: const InputDecoration(
                  labelText: 'Account Holder Name',
                  border: OutlineInputBorder(),
                ),
                validator: (value) =>
                    value == null || value.isEmpty ? 'Required' : null,
                onSaved: (val) => _formData['accountName'] = val!,
              ),
            ],
            const SizedBox(height: 24),
            SizedBox(
              width: double.infinity,
              child: ElevatedButton(
                onPressed: () {
                  if (_formKey.currentState!.validate()) {
                    _formKey.currentState!.save();
                    widget.onSubmit(_selectedType, _formData);
                  }
                },
                style: ElevatedButton.styleFrom(
                  padding: const EdgeInsets.symmetric(vertical: 16),
                ),
                child: const Text('Save Method'),
              ),
            ),
            const SizedBox(height: 24),
          ],
        ),
      ),
    );
  }
}
